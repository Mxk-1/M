> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate the A-share trading system frontend from a single `index.html` to a Vue 3 + Vite project with Pinia state management, preserving all existing features.

**Architecture:** The existing `index.html` JavaScript logic is extracted into focused Vue 3 SFCs using Composition API (`<script setup>`). All API calls are centralized in a composable `useMarketData.js`. Theme and color mode state is managed by a Pinia store persisted to `localStorage`. The FastAPI backend is updated to serve the compiled `dist/` directory instead of `static/`.

**Tech Stack:** Vue 3 (Composition API), Vite 5, Pinia, Canvas 2D, pure CSS variables (no UI library)

```js
import { defineStore } from 'pinia'
import { ref, watch } from 'vue'

export const useSettingsStore = defineStore('settings', () => {
  const THEMES = ['auto', 'light', 'dark']
  const THEME_ICONS = { auto: '🌗', light: '☀️', dark: '🌙' }
  const THEME_LABELS = { auto: '自动', light: '浅色', dark: '深色' }

  const savedTheme = localStorage.getItem('theme') || 'auto'
  const savedColor = localStorage.getItem('colorMode') || 'gr'

  const theme = ref(savedTheme)
  const colorMode = ref(savedColor)

  function applyTheme(t) {
    const h = document.documentElement
    h.setAttribute('data-theme', t)
    if (t === 'dark') h.classList.add('dark-mode')
    else h.classList.remove('dark-mode')
  }

  function applyColor(mode) {
    document.documentElement.setAttribute('data-color', mode)
  }

  function cycleTheme() {
    const idx = THEMES.indexOf(theme.value)
    theme.value = THEMES[(idx + 1) % THEMES.length]
    localStorage.setItem('theme', theme.value)
  }

  function toggleColor() {
    colorMode.value = colorMode.value === 'gr' ? 'rg' : 'gr'
    localStorage.setItem('colorMode', colorMode.value)
  }

  // Sync to DOM whenever values change
  watch(theme, (t) => applyTheme(t), { immediate: true })
  watch(colorMode, (m) => applyColor(m), { immediate: true })

  return {
    theme,
    colorMode,
    THEME_ICONS,
    THEME_LABELS,
    cycleTheme,
    toggleColor,
  }
})
```

```js
import { ref, computed } from 'vue'

export function useMarketData() {
  const dates = ref([])
  const currentDate = ref(null)
  const dateLabel = ref('加载中...')

  const indices = ref([])
  const sentiment = ref({})
  const sectors = ref([])
  const allSignals = ref([])
  const allMarket = ref([])

  const signalCounts = computed(() => {
    const counts = {}
    allSignals.value.forEach(s => {
      counts[s.strategy] = (counts[s.strategy] || 0) + 1
    })
    return counts
  })

  async function init() {
    const datesData = await fetch('/api/dates').then(r => r.json())
    dates.value = datesData
    currentDate.value = datesData[0]

    const d = String(currentDate.value)
    dateLabel.value = `${d.slice(0, 4)}-${d.slice(4, 6)}-${d.slice(6)} 盘后`

    const [market, sectorData, signals, stocks] = await Promise.all([
      fetch(`/api/market/${currentDate.value}`).then(r => r.json()),
      fetch(`/api/sectors/${currentDate.value}`).then(r => r.json()),
      fetch(`/api/signals/${currentDate.value}`).then(r => r.json()),
      fetch(`/api/stocks/${currentDate.value}`).then(r => r.json()),
    ])

    indices.value = market.indices
    sentiment.value = market.sentiment
    sectors.value = sectorData
    allSignals.value = signals
    allMarket.value = stocks
  }

  async function fetchKline(tsCode, days) {
    return fetch(`/api/kline/${tsCode}?days=${days}`).then(r => r.json())
  }

  return {
    dates,
    currentDate,
    dateLabel,
    indices,
    sentiment,
    sectors,
    allSignals,
    allMarket,
    signalCounts,
    init,
    fetchKline,
  }
}
```

```vue
<script setup>
import { ref, computed } from 'vue'

const props = defineProps({
  stocks: { type: Array, default: () => [] }
})

const emit = defineEmits(['open-kline'])

const searchVal = ref('')
const mktFilter = ref('ALL')
const sortState = ref({ col: 'pct_chg', dir: -1 })

function setFilter(f) {
  mktFilter.value = f
}

function setSort(col) {
  if (sortState.value.col === col) {
    sortState.value.dir *= -1
  } else {
    sortState.value.col = col
    sortState.value.dir = -1
  }
}

function sortClass(col) {
  if (sortState.value.col !== col) return ''
  return sortState.value.dir === 1 ? 'sort-asc' : 'sort-desc'
}

const filteredRows = computed(() => {
  let rows = [...props.stocks]
  const q = searchVal.value.trim().toLowerCase()
  if (q) rows = rows.filter(r => r.name.includes(q) || r.ts_code.toLowerCase().includes(q))

  const f = mktFilter.value
  if (f === 'UP')    rows = rows.filter(r => r.pct_chg > 0)
  if (f === 'DOWN')  rows = rows.filter(r => r.pct_chg < 0)
  if (f === 'LIMIT') rows = rows.filter(r => r.pct_chg >= 9.5)
  if (f === 'SH')    rows = rows.filter(r => r.ts_code.endsWith('.SH'))
  if (f === 'SZ')    rows = rows.filter(r => r.ts_code.endsWith('.SZ'))

  const { col, dir } = sortState.value
  rows.sort((a, b) => (a[col] > b[col] ? 1 : -1) * dir)
  return rows
})

function fmtAmt(v) {
  if (v >= 1e8) return `${(v / 1e8).toFixed(1)}亿`
  if (v >= 1e4) return `${(v / 1e4).toFixed(0)}万`
  return String(v)
}

function rowClass(pct) {
  if (pct > 0) return 'up'
  if (pct < 0) return 'dn'
  return 'flat'
}
</script>

<template>
  <div class="pane active" style="display:flex;flex:1;overflow:hidden;flex-direction:column">
    <div class="toolbar">
      <input
        class="search-input"
        v-model="searchVal"
        placeholder="搜索股票名称或代码..."
      />
      <div style="display:flex;gap:6px;flex-wrap:wrap">
        <button
          v-for="f in ['ALL','UP','DOWN','LIMIT','SH','SZ']"
          :key="f"
          class="filter-btn"
          :class="{ active: mktFilter === f }"
          @click="setFilter(f)"
        >{{ { ALL:'全部', UP:'上涨', DOWN:'下跌', LIMIT:'涨停', SH:'上证', SZ:'深证' }[f] }}</button>
      </div>
      <span class="toolbar-spacer"></span>
      <span class="count-label">{{ filteredRows.length }} 只</span>
    </div>
    <div class="table-wrap">
      <table>
        <thead>
          <tr>
            <th>#</th>
            <th :class="sortClass('name')" @click="setSort('name')">股票</th>
            <th :class="sortClass('pct_chg')" @click="setSort('pct_chg')">涨幅</th>
            <th :class="sortClass('close')" @click="setSort('close')">收盘</th>
            <th :class="sortClass('amount')" @click="setSort('amount')">成交额</th>
            <th>行业</th>
          </tr>
        </thead>
        <tbody>
          <tr v-if="filteredRows.length === 0">
            <td colspan="6" class="loading">暂无数据</td>
          </tr>
          <tr
            v-for="(r, i) in filteredRows"
            :key="r.ts_code"
            @click="emit('open-kline', r.ts_code, r.name)"
          >
            <td style="color:var(--text-2);font-size:12px">{{ i + 1 }}</td>
            <td>
              <div class="sn">{{ r.name }}</div>
              <div class="sc">{{ r.ts_code }}</div>
            </td>
            <td :class="rowClass(r.pct_chg)" style="font-weight:600;font-variant-numeric:tabular-nums">
              {{ r.pct_chg > 0 ? '+' : '' }}{{ r.pct_chg.toFixed(2) }}%
            </td>
            <td style="font-variant-numeric:tabular-nums">{{ r.close.toFixed(2) }}</td>
            <td style="color:var(--text-2);font-variant-numeric:tabular-nums">{{ fmtAmt(r.amount) }}</td>
            <td style="color:var(--text-2);font-size:12px">{{ r.industry || '—' }}</td>
          </tr>
        </tbody>
      </table>
    </div>
  </div>
</template>
```

```vue
function isDarkNow() {
  const t = document.documentElement.getAttribute('data-theme')
  return t === 'dark' || (t === 'auto' && window.matchMedia('(prefers-color-scheme:dark)').matches)

function drawKline() {
  const wrap = chartWrap.value
  if (!wrap) return
  const W = wrap.clientWidth
  const H = wrap.clientHeight
  if (!W || !H) { requestAnimationFrame(drawKline); return }

  const dpr = window.devicePixelRatio || 1
  const canvases = [cvBase.value, cvCross.value]
  canvases.forEach(c => {
    c.width = W * dpr
    c.height = H * dpr
    c.style.width = W + 'px'
    c.style.height = H + 'px'
    c.getContext('2d').scale(dpr, dpr)
  })

  const data = klineData.value
  if (!data.length) return

  const isDark = isDarkNow()
  const PAD = { top: 24, right: 60, bottom: 56, left: 8 }
  const VPAD = 0.06
  const VOL_H = Math.floor(H * 0.2)
  const priceH = H - PAD.top - PAD.bottom - VOL_H - 8
  const chartW = W - PAD.left - PAD.right

  const highs = data.map(d => d.high)
  const lows = data.map(d => d.low)
  const maxP = Math.max(...highs)
  const minP = Math.min(...lows)
  const range = maxP - minP || 1
  const padP = range * VPAD
  const pHigh = maxP + padP
  const pLow = minP - padP
  const maxVol = Math.max(...data.map(d => d.vol)) || 1

  const ctx = cvBase.value.getContext('2d')
  ctx.clearRect(0, 0, W, H)

  const cs = getComputedStyle(document.documentElement)
  const C = {
    up: cs.getPropertyValue('--up').trim() || '#30d158',
    dn: cs.getPropertyValue('--dn').trim() || '#ff453a',
    grid: isDark ? 'rgba(255,255,255,.06)' : 'rgba(0,0,0,.06)',
    text: isDark ? '#636366' : '#8e8e93',
    bg: isDark ? '#1c1c1e' : '#ffffff',
  }

  const toX = i => PAD.left + chartW * (i + 0.5) / data.length
  const toY = p => PAD.top + priceH * (1 - (p - pLow) / (pHigh - pLow))
  const toVolY = v => H - PAD.bottom - VOL_H * (v / maxVol)

  // Background
  ctx.fillStyle = C.bg
  ctx.fillRect(0, 0, W, H)

  // Horizontal grid + price labels (5 lines)
  ctx.font = '10px -apple-system,sans-serif'
  ctx.fillStyle = C.text
  ctx.textAlign = 'right'
  for (let i = 0; i <= 4; i++) {
    const p = pLow + (pHigh - pLow) * i / 4
    const y = toY(p)
    ctx.strokeStyle = C.grid
    ctx.lineWidth = 1
    ctx.beginPath(); ctx.moveTo(PAD.left, y); ctx.lineTo(W - PAD.right, y); ctx.stroke()
    ctx.fillText(p.toFixed(2), W - 4, y + 3.5)
  }

  // Vertical grid + date labels
  const step = Math.max(1, Math.floor(data.length / 6))
  for (let i = 0; i < data.length; i += step) {
    const x = toX(i)
    ctx.strokeStyle = C.grid
    ctx.lineWidth = 1
    ctx.beginPath(); ctx.moveTo(x, PAD.top); ctx.lineTo(x, H - PAD.bottom); ctx.stroke()
    const ds = String(data[i].date)
    ctx.fillStyle = C.text
    ctx.textAlign = 'center'
    ctx.fillText(`${ds.slice(4, 6)}/${ds.slice(6)}`, x, H - PAD.bottom + 14)
  }

  const cw = Math.max(1, Math.floor(chartW / data.length * 0.7))

  data.forEach((d, i) => {
    const x = toX(i)
    const isUp = d.close >= d.open
    const col = isUp ? C.up : C.dn
    ctx.strokeStyle = col
    ctx.fillStyle = col
    ctx.lineWidth = 1
    ctx.beginPath()
    ctx.moveTo(x, toY(d.high))
    ctx.lineTo(x, toY(d.low))
    ctx.stroke()

    const yO = toY(d.open)
    const yC = toY(d.close)
    const rectY = Math.min(yO, yC)
    const rectH = Math.max(1, Math.abs(yO - yC))
    if (isUp) ctx.strokeRect(x - cw / 2, rectY, cw, rectH)
    else ctx.fillRect(x - cw / 2, rectY, cw, rectH)

    ctx.fillStyle = isUp ? 'rgba(48,209,88,.5)' : 'rgba(255,69,58,.5)'
    const vy = toVolY(d.vol)
    ctx.fillRect(x - cw / 2, vy, cw, H - PAD.bottom - vy)
  })

  // Volume separator line
  ctx.strokeStyle = isDark ? 'rgba(255,255,255,.08)' : 'rgba(0,0,0,.08)'
  ctx.lineWidth = 1
  ctx.beginPath()
  ctx.moveTo(PAD.left, H - PAD.bottom - VOL_H)
  ctx.lineTo(W - PAD.right, H - PAD.bottom - VOL_H)
  ctx.stroke()

  // Crosshair layer
  const cv2 = cvCross.value
  cv2.onmousemove = e => drawCross(e, data, W, H, PAD, priceH, VOL_H, pHigh, pLow, maxVol, isDark, toX, toY, toVolY, chartW)
  cv2.onmouseleave = () => {
    cvCross.value.getContext('2d').clearRect(0, 0, W * dpr, H * dpr)
    klineTip.value = ''
  }
  klineTip.value = ''
}

function drawCross(e, data, W, H, PAD, priceH, VOL_H, pHigh, pLow, maxVol, isDark, toX, toY, toVolY, chartW) {
  const rect = e.target.getBoundingClientRect()
  const mx = e.clientX - rect.left
  const my = e.clientY - rect.top
  const idx = Math.round((mx - PAD.left) / chartW * data.length - 0.5)
  if (idx < 0 || idx >= data.length) return
  const d = data[idx]
  const x = toX(idx)

  const cv2 = cvCross.value
  const ctx = cv2.getContext('2d')
  const dpr = window.devicePixelRatio || 1
  ctx.clearRect(0, 0, W * dpr, H * dpr)

  const lineCol = isDark ? 'rgba(255,255,255,.3)' : 'rgba(0,0,0,.25)'
  ctx.strokeStyle = lineCol
  ctx.lineWidth = 1
  ctx.setLineDash([3, 3])
  ctx.beginPath(); ctx.moveTo(x, PAD.top); ctx.lineTo(x, H - PAD.bottom); ctx.stroke()
  if (my > PAD.top && my < H - PAD.bottom) {
    ctx.beginPath(); ctx.moveTo(PAD.left, my); ctx.lineTo(W - PAD.right, my); ctx.stroke()
  }
  ctx.setLineDash([])

  const price = pLow + (pHigh - pLow) * (1 - (my - PAD.top) / priceH)
  if (my > PAD.top && my < PAD.top + priceH) {
    ctx.fillStyle = isDark ? '#3a3a3c' : '#e5e5ea'
    ctx.fillRect(W - PAD.right, my - 9, PAD.right - 2, 18)
    ctx.fillStyle = isDark ? '#f5f5f7' : '#1d1d1f'
    ctx.font = '10px -apple-system,sans-serif'
    ctx.textAlign = 'right'
    ctx.fillText(price.toFixed(2), W - 4, my + 3.5)
  }

  const ds = String(d.date)
  const fmt = v => v >= 1e8 ? `${(v / 1e8).toFixed(1)}亿` : v >= 1e4 ? `${(v / 1e4).toFixed(0)}万` : v
  const sign = d.pct_chg >= 0 ? '+' : ''
  klineTip.value = `${ds.slice(0, 4)}-${ds.slice(4, 6)}-${ds.slice(6)}  开 ${d.open.toFixed(2)}  高 ${d.high.toFixed(2)}  低 ${d.low.toFixed(2)}  收 ${d.close.toFixed(2)}  ${sign}${d.pct_chg.toFixed(2)}%  量 ${fmt(d.vol)}`
}
```

```python
from pathlib import Path
import duckdb
from fastapi import FastAPI
from fastapi.staticfiles import StaticFiles
from fastapi.responses import FileResponse
from a_share_system.web.api import market, signals


def create_app(con: duckdb.DuckDBPyConnection) -> FastAPI:
    app = FastAPI(title="A股交易系统")
    app.include_router(market.router)
    app.include_router(signals.router)

    static_dir = Path(__file__).parent / "frontend" / "dist"
    static_dir.mkdir(parents=True, exist_ok=True)
    assets_dir = static_dir / "assets"
    if assets_dir.exists():
        app.mount("/assets", StaticFiles(directory=str(assets_dir)), name="assets")

    @app.get("/")
    def index():
        return FileResponse(str(static_dir / "index.html"))

    @app.get("/favicon.ico", include_in_schema=False)
    def favicon():
        favicon_path = static_dir / "favicon.ico"
        if not favicon_path.exists():
            favicon_path = Path(__file__).parent / "static" / "favicon.ico"
        return FileResponse(str(favicon_path))

    return app


if __name__ == "__main__":
    import uvicorn
    from a_share_system.data.db import get_conn
    con = get_conn()
    app = create_app(con)
    uvicorn.run(app, host="0.0.0.0", port=8080)
```
